```typescript
import * as lockfile from 'proper-lockfile'
import { getCredential } from './credential-store.js'
import {
  getWorkspaces,
  setWorkspaces,
  getWorkspaceForDomain,
  getConfigPath,
  type WorkspaceEntry,
} from './workspace-config.js'

export const REFRESH_THRESHOLD_MS = 5 * 60 * 1000 // 5 minutes

/**
 * Fetch workspace tokens from /api/2/user/tokens?cross_sites_p=1.
 * Uses the login token (from keychain) as the Authorization header.
 * Returns an array of WorkspaceEntry objects parsed from the API response.
 *
 * IMPORTANT: The response field names MUST match what was confirmed in Wave 0
 * (see 02-RESEARCH.md Pattern 6 — CONFIRMED section). Read that file before
 * implementing to get the exact field mapping.
 */
export async function fetchWorkspaceTokens(
  domain: string,
  loginToken: string
): Promise<WorkspaceEntry[]> {
  const url = `https://${domain}/api/2/user/tokens?cross_sites_p=1`
  const response = await fetch(url, {
    headers: { Authorization: `Bearer ${loginToken}` },
  })
  if (!response.ok) {
    throw new Error(`Failed to fetch workspace tokens: ${response.status} ${response.statusText}`)
  }
  const json = await response.json()
  // Map the API response to WorkspaceEntry[] using the confirmed field names
  // from 02-RESEARCH.md Pattern 6. The exact mapping depends on what Wave 0 confirmed.
  // Placeholder mapping — executor MUST read 02-RESEARCH.md for confirmed field names:
  const sites = Array.isArray(json) ? json : (json.sites ?? json.data ?? [])
  return sites.map((site: Record<string, unknown>) => ({
    domain: (site.domain ?? site.site_domain ?? site.hostname ?? '') as string,
    displayName: (site.name ?? site.display_name ?? site.site_name ?? site.domain ?? '') as string,
    token: (site.token ?? site.access_token ?? '') as string,
    expiresAt: typeof site.expires_at === 'number'
      ? site.expires_at * 1000 // convert seconds to ms if needed
      : (site.expire_time as number ?? 0),
  }))
}

/**
 * Ensure a fresh workspace token for the given domain.
 * Returns the token string, or null if no token exists.
 *
 * If the token is near expiry (within REFRESH_THRESHOLD_MS), acquires a
 * file lock and refreshes by re-calling /api/2/user/tokens.
 */
export async function ensureFreshToken(domain: string): Promise<string | null> {
  const workspace = getWorkspaceForDomain(domain)
  if (!workspace?.token) return null

  const now = Date.now()
  if (workspace.expiresAt - now > REFRESH_THRESHOLD_MS) {
    return workspace.token // still valid
  }

  // Need to refresh — acquire lock first
  const configPath = getConfigPath()
  let release: (() => Promise<void>) | undefined
  try {
    release = await lockfile.lock(configPath, { realpath: false, retries: 3 })

    // Re-check after lock (another process may have refreshed)
    const rechecked = getWorkspaceForDomain(domain)
    if (rechecked && rechecked.expiresAt - Date.now() > REFRESH_THRESHOLD_MS) {
      return rechecked.token
    }

    // Fetch fresh tokens using the login token from keychain
    const loginToken = getCredential(domain)
    if (!loginToken) return null

    const freshTokens = await fetchWorkspaceTokens(domain, loginToken)

    // Merge fresh tokens into existing workspace list
    const existing = getWorkspaces()
    const updated = existing.map(w => {
      const fresh = freshTokens.find(f => f.domain === w.domain)
      return fresh ? { ...w, token: fresh.token, expiresAt: fresh.expiresAt } : w
    })
    setWorkspaces(updated)

    return freshTokens.find(t => t.domain === domain)?.token ?? null
  } finally {
    await release?.()
  }
}
```
